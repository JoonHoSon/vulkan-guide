find_package(Vulkan REQUIRED)

include(FetchContent)

set(VENDOR_DIR "${CMAKE_CURRENT_SOURCE_DIR}")

# ----------------------------------------------------------------------------------------------------------------------
# GLM
# ----------------------------------------------------------------------------------------------------------------------
find_package(GLM QUIET)

if (NOT GLM_FOUND)
    message("GLM download from github repository...")

    FetchContent_Declare(GLM
        GIT_REPOSITORY https://github.com/g-truc/glm.git
        GIT_TAG refs/tags/1.0.3
        GIT_SHALLOW TRUE
        GIT_PROGRESS TRUE
        SOURCE_DIR "${VENDOR_DIR}/glm"
    )
    FetchContent_MakeAvailable(GLM)
else ()
    message("Using GLM from system library.")
endif ()

# ----------------------------------------------------------------------------------------------------------------------
# SDL
#   - SDL3 3.4.4
#   - SDL_image 3.4.2
# ----------------------------------------------------------------------------------------------------------------------
find_package(SDL3 QUIET)

if (NOT SDL3_FOUND)
    message("SDL3 download from github repository...")

    FetchContent_Declare(SDL3
        GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
        GIT_TAG refs/tags/release-3.4.4
        GIT_SHALLOW TRUE
        GIT_PROGRESS TRUE
        SOURCE_DIR "${VENDOR_DIR}/sdl3"
    )
    FetchContent_MakeAvailable(SDL3)
else ()
    message("Using SDL3 from system library.")
endif ()
